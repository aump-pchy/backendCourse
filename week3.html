Update & Delete

ในระบบยืมคืนจริง จำนวนอุปกรณ์เปลี่ยนทุกวัน และบางชิ้นต้องเลิกใช้ สัปดาห์นี้เราเติมการแก้ไขและการลบ API ของเราจะทำได้ครบ CRUD

ทำงานนี้แล้วจะได้อะไร

  • แก้ข้อมูลเฉพาะส่วนที่เปลี่ยนด้วย PATCH โดยไม่ต้องส่งข้อมูลทั้งหมด
  • เขียน Validation ที่ไม่ปล่อยให้ข้อมูลเปลี่ยนไปครึ่งเดียวเมื่อเจอค่าผิด
  • ลบข้อมูลได้ถูกตัว ไม่ลบผิดชิ้นโดยไม่รู้ตัว
  • ออกแบบชุดทดสอบ API ตามลำดับและพิสูจน์ผลด้วยหลักฐาน
เป้าหมายแนวคิดลงมือเขียนทดสอบตรวจผลลองเองส่งงาน

2. CRUD และ PATCH

งานเกือบทุกระบบมี 4 อย่างนี้ เรียกรวมว่า CRUD

CRUDMethodEndpoint ของเราสถานะ
Create สร้างPOST/api/equipmentWeek 2
Read อ่านGET/api/equipment, /:idWeek 1 และ 2
Update แก้ไขPATCH/api/equipment/:idสัปดาห์นี้

PATCH ต่างจาก POST และ PUT อย่างไร

Methodทำอะไรส่งข้อมูลแค่ไหนตัวอย่าง
POSTสร้างรายการใหม่ครบทุกฟิลด์ที่จำเป็นรับ Speaker เข้าระบบ
PUTแทนที่รายการเดิมทั้งชิ้นครบทุกฟิลด์กรอกข้อมูลอุปกรณ์ใหม่ทั้งหมด
PATCHแก้บางส่วนของรายการเดิมเฉพาะฟิลด์ที่เปลี่ยนนับของแล้วเหลือ 8 แก้แค่ quantity

คาบนี้ลงมือทำ PATCH เพราะงานจริงส่วนใหญ่แก้ทีละไม่กี่ฟิลด์ ส่วน PUT ให้รู้แนวคิดไว้ ถ้าส่ง PUT มาแค่ {"quantity":8} ตามหลักแล้ว name กับ category จะหายไปด้วย เพราะ PUT คือการแทนที่ทั้งชิ้น

  • ต้องการแก้ quantity ของ Notebook เป็น 8 อย่างเดียว ควรใช้แบบใด

ลงมือเขียน

3. Workshop PATCH

กติกาของ PATCH ในงานนี้

  • แก้ได้เฉพาะ name, category, quantity ต้องส่งมาอย่างน้อย 1 ฟิลด์
  • ปฏิเสธฟิลด์อื่นทั้งหมด รวมถึงการพยายามแก้ id
  • Body ต้องเป็น JSON Object ไม่ใช่ Array หรือ null
  • ตรวจเฉพาะฟิลด์ที่ส่งมา และยอมรับ quantity: 0
  • ตรวจทุกฟิลด์ให้ผ่านก่อน แล้วจึงเปลี่ยนข้อมูล ฟิลด์ที่ไม่ได้ส่งต้องคงค่าเดิม
  • ID ผิดรูปแบบตอบ 400 และ ID ที่ไม่มีตอบ 404

กับดัก 1: if (quantity)

อยากรู้ว่า Client ส่ง quantity มาไหม หลายคนเขียน if (quantity) แต่ 0 ใน JavaScript ถือเป็นค่าเท็จ ถ้าอุปกรณ์ถูกยืมไปหมดแล้วส่ง {"quantity":0} มา เงื่อนไขจะเป็นเท็จ ระบบจะเข้าใจว่าไม่ได้ส่งมา แล้วไม่แก้ค่า

// ผิด: 0 ถูกมองว่าไม่ได้ส่งมา
if (body.quantity) { ... }

// ถูก: ถามว่ามีฟิลด์ชื่อ quantity อยู่ใน Body หรือไม่
if (Object.hasOwn(body, 'quantity')) { ... }

กับดัก 2: แก้ไปครึ่งเดียว

ถ้าตรวจไปแก้ไป เช่น แก้ name ก่อน แล้วค่อยเจอว่า quantity ติดลบ จะตอบ 400 ก็จริง แต่ name เปลี่ยนไปแล้ว ข้อมูลในระบบจึงผิด ต้องแยกเป็น 2 ขั้น: ตรวจทั้งหมดก่อน ผ่านครบแล้ว ค่อยเปลี่ยน

// ผิด: ตรวจไปแก้ไป
if (Object.hasOwn(body, 'name')) item.name = body.name.trim();
if (!isValidQuantity(body.quantity)) return res.status(400)...  // name เปลี่ยนไปแล้ว

โค้ด PATCH

เพิ่มรายการฟิลด์ที่อนุญาตไว้ใต้ let nextId = 104;

server.js
const ALLOWED_FIELDS = ['name', 'category', 'quantity'];

เพิ่ม Route PATCH ต่อจาก Route POST และอยู่ก่อน Error handler

server.js
app.patch('/api/equipment/:id', (req, res) => {
  const id = parseId(req.params.id);
  if (id === null) {
    return res.status(400).json({ message: 'ID ต้องเป็นจำนวนเต็มบวก' });
  }

  const item = equipment.find(e => e.id === id);
  if (!item) {
    return res.status(404).json({ message: `ไม่พบอุปกรณ์ ID ${id}` });
  }

  const body = req.body;
  if (!isPlainObject(body)) {
    return res.status(400).json({ message: 'Body ต้องเป็น JSON Object' });
  }

  const fields = Object.keys(body);
  if (fields.length === 0) {
    return res.status(400).json({ message: 'ต้องส่งอย่างน้อย 1 ฟิลด์: name, category หรือ quantity' });
  }

  const notAllowed = fields.filter(f => !ALLOWED_FIELDS.includes(f));
  if (notAllowed.length > 0) {
    return res.status(400).json({ message: `ไม่อนุญาตให้แก้ฟิลด์: ${notAllowed.join(', ')}` });
  }

  // ขั้นที่ 1: ตรวจทุกฟิลด์ที่ส่งมา
  if (Object.hasOwn(body, 'name') && !isNonEmptyString(body.name)) {
    return res.status(400).json({ message: 'name ต้องเป็นข้อความที่ไม่ว่าง' });
  }
  if (Object.hasOwn(body, 'category') && !isNonEmptyString(body.category)) {
    return res.status(400).json({ message: 'category ต้องเป็นข้อความที่ไม่ว่าง' });
  }
  if (Object.hasOwn(body, 'quantity') && !isValidQuantity(body.quantity)) {
    return res.status(400).json({ message: 'quantity ต้องเป็นจำนวนเต็มตั้งแต่ 0 ขึ้นไป' });
  }

  // ขั้นที่ 2: ผ่านครบแล้วจึงเปลี่ยนข้อมูล
  if (Object.hasOwn(body, 'name')) item.name = body.name.trim();
  if (Object.hasOwn(body, 'category')) item.category = body.category.trim();
  if (Object.hasOwn(body, 'quantity')) item.quantity = body.quantity;

  res.json(item);
});

Object.keys(body) ให้รายชื่อฟิลด์ที่ส่งมาทั้งหมด เช่น ['quantity'] เราใช้ตรวจว่าว่างไหม และมีฟิลด์นอกรายการ ALLOWED_FIELDS หรือไม่ ถ้าส่ง {"id":999} มาจะติดตรงนี้ ID จึงแก้ไม่ได้

Body ที่ส่งผลที่คาดหวังสิ่งที่ทดสอบ

{"quantity":0}200 quantity เป็น 0ยอมรับ 0
{"name":"Notebook Pro","quantity":-1}400 แล้ว GET ดูต้องยังชื่อเดิมไม่แก้ครึ่งเดียว

5. Workshop DELETE

การลบใช้ 2 คำสั่งของ Array คือ findIndex() หาตำแหน่ง แล้ว splice(ตำแหน่ง, 1) ตัดออก 1 รายการ แต่ถ้าหาไม่เจอ findIndex() จะคืนค่า -1 และ splice(-1, 1) หมายถึงลบตัวสุดท้ายของ Array

ลองลบแบบตรวจและไม่ตรวจ -1

ใส่ ID 9999 แล้วกดลบแบบไม่ตรวจ ดูว่าตัวไหนหายไป

เพิ่ม Route DELETE ต่อจาก PATCH

server.js
app.delete('/api/equipment/:id', (req, res) => {
  const id = parseId(req.params.id);
  if (id === null) {
    return res.status(400).json({ message: 'ID ต้องเป็นจำนวนเต็มบวก' });
  }

  const index = equipment.findIndex(e => e.id === id);
  if (index === -1) {
    return res.status(404).json({ message: `ไม่พบอุปกรณ์ ID ${id}` });
  }

  const [deleted] = equipment.splice(index, 1);
  res.json({ message: `ลบอุปกรณ์ ID ${id} แล้ว`, deleted });
});

splice() คืนค่าเป็น Array ของรายการที่ถูกตัดออก const [deleted] = ... ดึงตัวแรกออกมา เพื่อตอบกลับให้ Client เห็นว่าลบอะไรไป

ไม่เรียง ID ใหม่ ไม่ลด nextId

ลบ 102 แล้ว ID ที่เหลือคือ 101 กับ 103 ปล่อยให้เว้นช่องไว้แบบนั้น และ nextId ยังเป็นค่าเดิม เพราะ ID ใช้ระบุตัวตน ถ้าเรียงใหม่ ใบยืมที่อ้าง ID 103 ไว้จะชี้ไปผิดชิ้น

เฉาก๊วยสงสัย

ประกาศ equipment ด้วย const แล้วทำไมยัง splice() ได้ ? เพราะ const แค่ห้ามเอา Array ใหม่มาแทน แต่ไม่ได้ห้ามเพิ่มหรือลบของข้างใน

ดู server.js ฉบับเต็มของ Week 3
server.js
import express from 'express';

const app = express();
const PORT = 3000;

app.use(express.json());

const equipment = [
  { id: 101, name: 'Notebook', category: 'computer', quantity: 5 },
  { id: 102, name: 'Projector', category: 'display', quantity: 2 },
  { id: 103, name: 'PC', category: 'computer', quantity: 10 }
];

let nextId = 104;

const ALLOWED_FIELDS = ['name', 'category', 'quantity'];

function isNonEmptyString(value) {
  return typeof value === 'string' && value.trim() !== '';
}

function parseId(value) {
  if (!/^\d+$/.test(value)) return null;
  const id = Number(value);
  return Number.isSafeInteger(id) && id > 0 ? id : null;
}

function isValidQuantity(value) {
  return Number.isSafeInteger(value) && value >= 0;
}

function isPlainObject(value) {
  return value !== null && typeof value === 'object' && !Array.isArray(value);
}

app.get('/', (req, res) => {
  res.send('Hello, Backend!');
});

app.get('/api/health', (req, res) => {
  res.json({ status: 'ok' });
});

app.get('/api/equipment', (req, res) => {
  const { category } = req.query;

  if (category === undefined) {
    return res.json(equipment);
  }

  if (!isNonEmptyString(category)) {
    return res.status(400).json({ message: 'category ต้องเป็นข้อความที่ไม่ว่าง' });
  }

  const keyword = category.trim().toLowerCase();
  const result = equipment.filter(item => item.category.toLowerCase() === keyword);
  res.json(result);
});

app.get('/api/equipment/count', (req, res) => {
  res.json({ total: equipment.length });
});

app.get('/api/equipment/:id', (req, res) => {
  const id = parseId(req.params.id);

  if (id === null) {
    return res.status(400).json({ message: 'ID ต้องเป็นจำนวนเต็มบวก' });
  }

  const item = equipment.find(e => e.id === id);

  if (!item) {
    return res.status(404).json({ message: `ไม่พบอุปกรณ์ ID ${id}` });
  }

  res.json(item);
});

app.post('/api/equipment', (req, res) => {
  const body = req.body;

  if (!isPlainObject(body)) {
    return res.status(400).json({ message: 'Body ต้องเป็น JSON Object' });
  }

  const { name, category, quantity } = body;

  if (!isNonEmptyString(name)) {
    return res.status(400).json({ message: 'name ต้องเป็นข้อความที่ไม่ว่าง' });
  }
  if (!isNonEmptyString(category)) {
    return res.status(400).json({ message: 'category ต้องเป็นข้อความที่ไม่ว่าง' });
  }
  if (!isValidQuantity(quantity)) {
    return res.status(400).json({ message: 'quantity ต้องเป็นจำนวนเต็มตั้งแต่ 0 ขึ้นไป' });
  }

  const newItem = {
    id: nextId++,
    name: name.trim(),
    category: category.trim(),
    quantity
  };

  equipment.push(newItem);
  res.status(201).json(newItem);
});

app.patch('/api/equipment/:id', (req, res) => {
  const id = parseId(req.params.id);
  if (id === null) {
    return res.status(400).json({ message: 'ID ต้องเป็นจำนวนเต็มบวก' });
  }

  const item = equipment.find(e => e.id === id);
  if (!item) {
    return res.status(404).json({ message: `ไม่พบอุปกรณ์ ID ${id}` });
  }

  const body = req.body;
  if (!isPlainObject(body)) {
    return res.status(400).json({ message: 'Body ต้องเป็น JSON Object' });
  }

  const fields = Object.keys(body);
  if (fields.length === 0) {
    return res.status(400).json({ message: 'ต้องส่งอย่างน้อย 1 ฟิลด์: name, category หรือ quantity' });
  }

  const notAllowed = fields.filter(f => !ALLOWED_FIELDS.includes(f));
  if (notAllowed.length > 0) {
    return res.status(400).json({ message: `ไม่อนุญาตให้แก้ฟิลด์: ${notAllowed.join(', ')}` });
  }

  // ขั้นที่ 1: ตรวจทุกฟิลด์ที่ส่งมา
  if (Object.hasOwn(body, 'name') && !isNonEmptyString(body.name)) {
    return res.status(400).json({ message: 'name ต้องเป็นข้อความที่ไม่ว่าง' });
  }
  if (Object.hasOwn(body, 'category') && !isNonEmptyString(body.category)) {
    return res.status(400).json({ message: 'category ต้องเป็นข้อความที่ไม่ว่าง' });
  }
  if (Object.hasOwn(body, 'quantity') && !isValidQuantity(body.quantity)) {
    return res.status(400).json({ message: 'quantity ต้องเป็นจำนวนเต็มตั้งแต่ 0 ขึ้นไป' });
  }

  // ขั้นที่ 2: ผ่านครบแล้วจึงเปลี่ยนข้อมูล
  if (Object.hasOwn(body, 'name')) item.name = body.name.trim();
  if (Object.hasOwn(body, 'category')) item.category = body.category.trim();
  if (Object.hasOwn(body, 'quantity')) item.quantity = body.quantity;

  res.json(item);
});

app.delete('/api/equipment/:id', (req, res) => {
  const id = parseId(req.params.id);
  if (id === null) {
    return res.status(400).json({ message: 'ID ต้องเป็นจำนวนเต็มบวก' });
  }

  const index = equipment.findIndex(e => e.id === id);
  if (index === -1) {
    return res.status(404).json({ message: `ไม่พบอุปกรณ์ ID ${id}` });
  }

  const [deleted] = equipment.splice(index, 1);
  res.json({ message: `ลบอุปกรณ์ ID ${id} แล้ว`, deleted });
});

app.use((err, req, res, next) => {
  if (err.type === 'entity.parse.failed') {
    return res.status(400).json({ message: 'รูปแบบ JSON ไม่ถูกต้อง' });
  }
  console.error(err);
  res.status(500).json({ message: 'เกิดข้อผิดพลาดที่ Server' });
});

app.listen(PORT, () => {
  console.log(`Server running at http://localhost:${PORT}`);
});

ตอบด้วยคำพูดของตัวเองสั้น ๆ แล้วกดสรุปเพื่อคัดลอกส่งครู

1. PATCH ต่างจาก POST อย่างไร และทำไมการแก้ quantity อย่างเดียวจึงไม่ต้องส่ง name?

แนวคำตอบ: POST สร้างรายการใหม่ ส่วน PATCH แก้บางส่วนของรายการที่มีอยู่ เราแก้เฉพาะฟิลด์ที่ส่งมา ฟิลด์ที่ไม่ได้ส่งจึงคงค่าเดิม

2. ทำไม if (quantity) จึงไม่เหมาะกับการตรวจข้อมูลในงานนี้?

แนวคำตอบ: 0 เป็นค่าเท็จใน JavaScript ถ้าส่ง quantity เป็น 0 มา เงื่อนไขจะเป็นเท็จเหมือนไม่ได้ส่ง ค่าจึงไม่ถูกแก้ ต้องตรวจว่ามีฟิลด์อยู่ด้วย Object.hasOwn

3. findIndex() คืน -1 หมายความว่าอย่างไร และต้องตรวจอะไรบ้างก่อน splice()?

แนวคำตอบ: -1 คือหาไม่พบ ต้องตรวจว่า ID รูปแบบถูก (ไม่ถูกตอบ 400) และ index ไม่เท่ากับ -1 (ถ้าเท่าตอบ 404) เพราะ splice(-1, 1) จะลบรายการสุดท้ายผิดตัว

4. เมื่อ DELETE สำเร็จแล้วลบ ID เดิมอีกครั้งควรตอบสถานะอะไร เพราะเหตุใด?

แนวคำตอบ: 404 เพราะ ID มีรูปแบบถูกต้อง แต่รายการนั้นถูกลบไปแล้วจึงไม่มีอยู่ในระบบ และจำนวนรายการต้องไม่ลดซ้ำ

5. ถ้าส่ง name ถูกต้องแต่ quantity ติดลบ ข้อมูลควรเปลี่ยนบางส่วนหรือไม่? เพราะเหตุใด?

แนวคำตอบ: ไม่ควรเปลี่ยนเลย ต้องตอบ 400 และคงข้อมูลเดิมทั้งหมด เพราะคำขอนี้ไม่ถูกต้องทั้งชุด ถ้าเปลี่ยนบางส่วน ข้อมูลในระบบจะไม่ตรงกับที่ผู้ใช้ตั้งใจ
